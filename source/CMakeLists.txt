cmake_minimum_required(VERSION 3.24)
project(Uprising VERSION 0.2.0 LANGUAGES C)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Uprising requires a 64-bit target (x64 or ARM64)")
endif()
if(WIN32)
    enable_language(RC)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(UPRISING_ASSET_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "Build output folder; local builds share game assets")
get_filename_component(UPRISING_ASSET_DIRECTORY "${UPRISING_ASSET_DIRECTORY}" ABSOLUTE)
include(cmake/sources.cmake)
include(cmake/sdl.cmake)
option(UPRISING_VULKAN_PROBE_ONLY "Build only the asset-free Vulkan/FSR3 probe" OFF)
if(UPRISING_VULKAN_PROBE_ONLY)
    include(cmake/vulkan_probe.cmake)
    return()
endif()
include(cmake/enet.cmake)
include(cmake/freetype.cmake)
include(cmake/vulkan.cmake)

foreach(path IN LISTS CORE_SOURCES GAME_SOURCES PLATFORM_SOURCES)
    if(path MATCHES "(^|/)(cpu|generated|overlay)/")
        message(FATAL_ERROR "Forbidden execution source: ${path}")
    endif()
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/${path}" source_text)
    if(source_text MATCHES "#[ \t]*include[ \t]*[\"<](cpu|x87)\\.h")
        message(FATAL_ERROR "CPU dependency in shipping source: ${path}")
    endif()
endforeach()

if(MSVC)
    set(native_options /O2 /fp:strict /Gy /W3)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
else()
    set(native_options -O2 -frounding-math -ffunction-sections -fdata-sections -Wall -Wextra)
endif()
add_library(uprising_core OBJECT ${CORE_SOURCES})
target_include_directories(uprising_core PUBLIC core platform)
target_compile_options(uprising_core PRIVATE ${native_options})
set(default_executable_name uprising)
if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(default_executable_name uprising64)
endif()
set(UPRISING_EXECUTABLE_NAME "${default_executable_name}" CACHE STRING "Output basename; CI uses uprising")
if(NOT UPRISING_EXECUTABLE_NAME MATCHES "^uprising(64)?$")
    message(FATAL_ERROR "Unsupported executable name")
endif()
configure_file(resources/version.h.in generated/version.h @ONLY)
if(MINGW)
    set(resource_target pe-x86-64)
    add_custom_command(
        OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/uprising.res" "${CMAKE_CURRENT_BINARY_DIR}/generated/uprising_resources.o"
        COMMAND "${CMAKE_RC_COMPILER}" --input-format=rc --output-format=res
            "-I${CMAKE_CURRENT_SOURCE_DIR}/resources" "-I${CMAKE_CURRENT_BINARY_DIR}/generated"
            "${CMAKE_CURRENT_SOURCE_DIR}/resources/uprising.rc" "${CMAKE_CURRENT_BINARY_DIR}/generated/uprising.res"
        COMMAND "${CMAKE_RC_COMPILER}" --input-format=res --output-format=coff --target=${resource_target}
            "${CMAKE_CURRENT_BINARY_DIR}/generated/uprising.res" "${CMAKE_CURRENT_BINARY_DIR}/generated/uprising_resources.o"
        DEPENDS resources/uprising.rc resources/uprising.ico resources/version.h.in "${CMAKE_CURRENT_BINARY_DIR}/generated/version.h"
        VERBATIM)
    set(native_resources "${CMAKE_CURRENT_BINARY_DIR}/generated/uprising_resources.o")
    set_source_files_properties(${native_resources} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
elseif(WIN32)
    set(native_resources resources/uprising.rc)
elseif(APPLE)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/uprising.icns"
        COMMAND /bin/bash "${CMAKE_CURRENT_SOURCE_DIR}/scripts/macos-icon.sh"
            "${CMAKE_CURRENT_SOURCE_DIR}/resources/uprising.ico" "${CMAKE_CURRENT_BINARY_DIR}/generated/uprising.icns"
        DEPENDS resources/uprising.ico scripts/macos-icon.sh VERBATIM)
    set(native_resources "${CMAKE_CURRENT_BINARY_DIR}/generated/uprising.icns")
    set_source_files_properties(${native_resources} PROPERTIES MACOSX_PACKAGE_LOCATION Resources GENERATED TRUE)
endif()
add_executable(uprising WIN32 ${GAME_SOURCES} ${PLATFORM_SOURCES} third_party/libsmacker/smacker.c ${native_resources})
set_target_properties(uprising PROPERTIES OUTPUT_NAME "${UPRISING_EXECUTABLE_NAME}")
target_include_directories(uprising PRIVATE core game platform resources "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(uprising PRIVATE ${native_options})
if(MSVC)
    target_link_options(uprising PRIVATE /OPT:REF)
elseif(APPLE)
    target_link_options(uprising PRIVATE -Wl,-dead_strip)
else()
    target_link_options(uprising PRIVATE -Wl,--gc-sections)
endif()
target_link_libraries(uprising PRIVATE uprising_core SDL3::SDL3-static enet freetype uprising_vulkan)
if(NOT MSVC)
    target_link_libraries(uprising PRIVATE m)
endif()
if(APPLE)
    set_target_properties(uprising PROPERTIES MACOSX_BUNDLE TRUE OUTPUT_NAME Uprising
        MACOSX_BUNDLE_BUNDLE_NAME Uprising MACOSX_BUNDLE_GUI_IDENTIFIER org.uprising.decomp
        MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
        MACOSX_BUNDLE_ICON_FILE uprising.icns)
endif()
set_target_properties(uprising PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${UPRISING_ASSET_DIRECTORY}")
foreach(configuration DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(uprising PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${configuration} "${UPRISING_ASSET_DIRECTORY}")
endforeach()

option(BUILD_TESTING "Build asset-free native checks" OFF)
if(BUILD_TESTING)
    enable_testing()
    include(cmake/vulkan_probe.cmake)
    add_executable(core_smoke tests/core_smoke.c)
    target_link_libraries(core_smoke PRIVATE uprising_core)
    add_test(NAME native_core COMMAND core_smoke)
    add_executable(config_test tests/config.c core/game_config.c)
    target_include_directories(config_test PRIVATE core)
    add_test(NAME configuration COMMAND config_test)
    add_executable(audio_test tests/audio.c)
    target_include_directories(audio_test PRIVATE core game platform)
    target_link_libraries(audio_test PRIVATE SDL3::SDL3-static)
    if(MSVC)
        target_compile_options(audio_test PRIVATE /UNDEBUG)
    else()
        target_link_libraries(core_smoke PRIVATE m)
        target_link_libraries(audio_test PRIVATE m)
        target_compile_options(audio_test PRIVATE -UNDEBUG)
    endif()
    add_test(NAME audio_mixer COMMAND audio_test)
endif()
include(cmake/install.cmake)
